    document.getElementById('refreshBtn').addEventListener('click', function(){ location.reload(); });
    const toggleBtn = document.getElementById('toggleBtn');
    const toggleAction = document.getElementById('toggleAction');
    const alertsState = document.getElementById('alertsState');
    if (alertsState && alertsState.innerText.trim() === 'Enabled'){
      toggleBtn.innerText = 'Disable Alerts';
      toggleAction.value = 'disable';
      toggleBtn.classList.remove('primary');
    } else {
      toggleBtn.innerText = 'Enable Alerts';
      toggleAction.value = 'enable';
      toggleBtn.classList.add('primary');
    }
    const live = document.getElementById('liveStream');
    let lastTime = Date.now();
    live.addEventListener('load', function(){
      const now = Date.now();
      const dt = now - lastTime;
      lastTime = now;
      const fps = (dt>0)?(1000/dt).toFixed(1):'--';
      document.getElementById('fps').innerText = 'FPS: ' + fps;
      document.getElementById('streamStatus').innerText = 'Live';
      fetch('/api/alerts_enabled').then(r => r.json()).then(j=>{ document.getElementById('alertsState').innerText = j.enabled ? 'Enabled' : 'Disabled'; }).catch(()=>{});
    });
    setInterval(()=>{ fetch(location.href).then(r=>r.text()).then(html=>{
      const doc = new DOMParser().parseFromString(html,'text/html');
      const newGallery = doc.getElementById('gallery');
      if (newGallery) document.getElementById('gallery').innerHTML = newGallery.innerHTML;
      const newLast = doc.getElementById('lastCap');
      if (newLast) document.getElementById('lastCap').innerText = newLast.innerText;
    }).catch(()=>{}); }, 10000);
